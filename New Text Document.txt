    body{font-family:Arial,sans-serif;max-width:480px;margin:0 auto;padding:20px}
    button{padding:12px 18px;font-size:16px;border:0;border-radius:6px;background:#1a73e8;color:#fff;width:100%}
    button:disabled{opacity:.6}
    input,textarea{width:100%;padding:10px;margin:6px 0 14px;box-sizing:border-box;font-size:16px}
    .msg{padding:10px;border-radius:6px;margin:12px 0}
    .err{background:#fde8e8;color:#b00020}.ok{background:#e6f4ea;color:#137333}
    .hidden{display:none}
  </style>

<body>
  <h2>Please fill this form</h2>

  <div id="gate">
    <p>We need your location before you can continue. Your location will be saved with your response.</p>
    <button id="locBtn">Allow location</button>
    <div id="gateMsg" class="msg err hidden"></div>
  </div>

  <form id="form" class="hidden">
    <div id="locInfo" class="msg ok"></div>
    <label>Name</label><input id="name" required>
    <label>Email</label><input id="email" type="email" required>
    <label>Comment</label><textarea id="comment" rows="3"></textarea>
    <button type="submit" id="sendBtn">Submit</button>
    <div id="formMsg" class="msg hidden"></div>
  </form>

<script>
const API_URL = 'https://script.google.com/macros/s/AKfycbyirCjoFzIwrAE61vH7FRZ_7C95G2l6G8nIHDrhNaGGkP7AKPVi81ukH9ml5RmIPO1s/exec';
let pos = null;
const $ = id => document.getElementById(id);

$('locBtn').onclick = () => {
  $('gateMsg').classList.add('hidden');
  if (!navigator.geolocation) return gateError('This browser does not support location.');
  $('locBtn').textContent = 'Getting your location...';
  navigator.geolocation.getCurrentPosition(onLoc, onErr,
    { enableHighAccuracy: true, timeout: 20000, maximumAge: 0 });
};

function onLoc(p) {
  pos = { lat: p.coords.latitude, lng: p.coords.longitude, accuracy: p.coords.accuracy };
  $('gate').classList.add('hidden');
  $('form').classList.remove('hidden');
  $('locInfo').textContent = 'Location captured (accuracy about ' + Math.round(pos.accuracy) + ' m).';
}

function onErr(e) {
  $('locBtn').textContent = 'Try again';
  const m = { 1: 'Location was denied. Allow location for this page in your browser settings, then try again.',
              2: 'Your location could not be determined. Turn on GPS and try again.',
              3: 'Getting your location timed out. Try again.' };
  gateError(m[e.code] || e.message);
}
function gateError(t) { $('gateMsg').textContent = t; $('gateMsg').classList.remove('hidden'); }

$('form').onsubmit = async ev => {
  ev.preventDefault();
  if (!pos) return;
  const btn = $('sendBtn'), msg = $('formMsg');
  btn.disabled = true; btn.textContent = 'Submitting...'; msg.classList.add('hidden');
  try {
    const res = await fetch(API_URL, {
      method: 'POST',   // no custom headers, so no CORS preflight
      body: JSON.stringify({
        name: $('name').value, email: $('email').value, comment: $('comment').value,
        lat: pos.lat, lng: pos.lng, accuracy: pos.accuracy })
    });
    const r = await res.json();
    if (!r.ok) throw new Error(r.error || 'Submission failed.');
    $('form').innerHTML = '<div class="msg ok">Thank you, your response has been recorded.</div>';
  } catch (err) {
    btn.disabled = false; btn.textContent = 'Submit';
    msg.className = 'msg err'; msg.textContent = err.message;
  }
};
</script>
</body>
